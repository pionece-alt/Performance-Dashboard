// @FROZEN: batch=3 date=2026-04-22 reason="Auth + ACCESS_LIST + inactivity 45 phút + auto-delete session"
function verifyFirebaseToken_(token) {
  try {
    var url = 'https://firestore.googleapis.com/v1/projects/bpd-ops/databases/(default)/documents/sessions/' + token;
    var serviceAccountKey = getServiceAccountKey_();
    var accessToken = getFirebaseAccessToken_(serviceAccountKey);
    var resp = UrlFetchApp.fetch(url, {
      headers: { 'Authorization': 'Bearer ' + accessToken },
      muteHttpExceptions: true
    });
    if (resp.getResponseCode() !== 200) return null;
    var doc = JSON.parse(resp.getContentText());
    var fields = doc.fields || {};
    var exp = fields.exp && fields.exp.integerValue ? parseInt(fields.exp.integerValue) : 0;
    if (exp < Date.now()) return null; // het han
    var email = fields.email && fields.email.stringValue ? fields.email.stringValue.trim().toLowerCase() : '';
    if (!email) return null;

    // Check email có trong ACCESS_LIST không — nếu đã bị xóa thì từ chối + xóa session luôn
    var allUsers = getUsers();
    var inList = allUsers.some(function(u){ return (u.email||'').trim().toLowerCase() === email; });
    if (!inList) {
      Logger.log('verifyFirebaseToken_: email không còn trong ACCESS_LIST: ' + email + ' — đang xóa session...');
      try {
        UrlFetchApp.fetch('https://firestore.googleapis.com/v1/projects/' + FIREBASE_PROJECT_ID + '/databases/(default)/documents/sessions/' + token, {
          method: 'DELETE',
          headers: { 'Authorization': 'Bearer ' + accessToken },
          muteHttpExceptions: true
        });
      } catch(de) {}
      return null;
    }

    // Option B: Check lastSeen — tab đóng > 45 phút = session chết
    // Bypass check nếu session mới (chưa từng có heartbeat) — bypass bằng cách skip nếu lastSeen chưa tồn tại
    var lastSeen = fields.lastSeen && (fields.lastSeen.integerValue || fields.lastSeen.doubleValue)
                   ? parseInt(fields.lastSeen.integerValue || fields.lastSeen.doubleValue) : 0;
    if (lastSeen && lastSeen < 1e12) lastSeen = lastSeen * 1000; // seconds → ms
    if (lastSeen) {
      var INACTIVITY_MAX_MS = 45 * 60 * 1000; // 45 phút
      var gap = Date.now() - lastSeen;
      if (gap > INACTIVITY_MAX_MS) {
        Logger.log('verifyFirebaseToken_: session expired by inactivity (' + Math.round(gap/60000) + ' min) — email: ' + email);
        try {
          UrlFetchApp.fetch('https://firestore.googleapis.com/v1/projects/' + FIREBASE_PROJECT_ID + '/databases/(default)/documents/sessions/' + token, {
            method: 'DELETE',
            headers: { 'Authorization': 'Bearer ' + accessToken },
            muteHttpExceptions: true
          });
        } catch(de) {}
        return null;
      }
    }
    // Note: Nếu lastSeen không tồn tại (session vừa login, chưa có heartbeat lần nào) → bypass check,
    // heartbeat đầu tiên sẽ update lastSeen trong 30s đầu tiên.

    return {
      email: email,
      role:  fields.role && fields.role.stringValue || 'Viewer'
    };
  } catch(e) {
    Logger.log('verifyFirebaseToken_ error: ' + e.message);
    return null;
  }
}

function getServiceAccountKey_() {
  var props = PropertiesService.getScriptProperties();
  var privateKey = props.getProperty('FIREBASE_PRIVATE_KEY') || '';
  privateKey = privateKey.replace(/\\n/g, '\n');
  return {
    client_email: props.getProperty('FIREBASE_CLIENT_EMAIL'),
    private_key: privateKey
  };
}

function getFirebaseAccessToken_(key) {
  var now = Math.floor(Date.now() / 1000);
  var header  = Utilities.base64EncodeWebSafe(JSON.stringify({ alg: 'RS256', typ: 'JWT' }));
  var payload = Utilities.base64EncodeWebSafe(JSON.stringify({
    iss: key.client_email, sub: key.client_email,
    aud: 'https://oauth2.googleapis.com/token',
    iat: now, exp: now + 3600,
    scope: 'https://www.googleapis.com/auth/datastore'
  }));
  var sig = Utilities.base64EncodeWebSafe(
    Utilities.computeRsaSha256Signature(header + '.' + payload, key.private_key)
  );
  var resp = UrlFetchApp.fetch('https://oauth2.googleapis.com/token', {
    method: 'POST',
    contentType: 'application/x-www-form-urlencoded',
    payload: 'grant_type=urn:ietf:params:oauth:grant-type:jwt-bearer&assertion=' + header + '.' + payload + '.' + sig,
    muteHttpExceptions: true
  });
  return JSON.parse(resp.getContentText()).access_token;
}

function loadAll(params) {
  try {
    var fbToken = (params && params.fbToken) || '';
    if (fbToken) {
      var sessUrl = 'https://firestore.googleapis.com/v1/projects/' + FIREBASE_PROJECT_ID +
                    '/databases/(default)/documents/sessions/' + fbToken;
      var fsToken = getFirebaseAccessToken_(getServiceAccountKey_());
      var sessResp = UrlFetchApp.fetch(sessUrl, {
        headers: { 'Authorization': 'Bearer ' + fsToken },
        muteHttpExceptions: true
      });
      if (sessResp.getResponseCode() !== 200) {
        return { ok: false, unauthorized: true };
      }
      var sessDoc = JSON.parse(sessResp.getContentText());
      var sessEmail = sessDoc.fields && sessDoc.fields.email ? sessDoc.fields.email.stringValue : '';
      if (sessEmail) {
        var allUsers = getUsers();
        var stillExists = allUsers.some(function(u){ return u.email === sessEmail; });
        if (!stillExists) return { ok: false, unauthorized: true };
      }
    }
    var pcMaxUsers = 0;
    try {
      var token = getFirebaseAccessToken_(getServiceAccountKey_());
      syncWebauthnFromFirestore_(token);
      var cfgUrl = 'https://firestore.googleapis.com/v1/projects/'+FIREBASE_PROJECT_ID+'/databases/(default)/documents/config/settings';
      var cfgResp = UrlFetchApp.fetch(cfgUrl,{headers:{'Authorization':'Bearer '+token},muteHttpExceptions:true});
      if(cfgResp.getResponseCode()===200){
        var cfgDoc = JSON.parse(cfgResp.getContentText());
        if(cfgDoc.fields&&cfgDoc.fields.max_users){
          pcMaxUsers=parseInt(cfgDoc.fields.max_users.integerValue||cfgDoc.fields.max_users.doubleValue||0)||0;
        }
      }
    } catch(fe){ Logger.log('loadAll firebase error: '+fe.message); }
    try { CacheService.getScriptCache().remove('bpd_users'); } catch(ce) {}
    var allUsers = getUsers();
    var curUser  = getCurrentUser_();
    var stillExists = allUsers.some(function(u){ return u.email === curUser.email; });
    if (!stillExists && curUser.email) {
      return { ok: false, unauthorized: true };
    }
    return {
      ok          : true,
      tierRules   : getTierRules(),
      appConfig   : getAppConfig(),
      users       : allUsers,
      brandLevels : getBrandLevels(),
      currentUser : curUser,
      unauthorized: false,
      maxUsers    : pcMaxUsers
    };
  } catch(e) {
    return { ok: false, error: e.message };
  }
}

function getCurrentUser_() {
  try {
    var email = Session.getActiveUser().getEmail();
    var users  = getUsers();
    var found  = users.find(function(u){ return u.email === email; });
    if (found) return { name: found.name, email: email, role: found.role, color: found.color, level: found.level||'', brands: found.brands||[] };
    return { name: email.split('@')[0], email: email, role: 'Viewer', color: '#94a3b8', level: '', brands: [] };
  } catch(e) {
    return { name: 'Guest', email: '', role: 'Viewer', color: '#94a3b8', level: '', brands: [] };
  }
}

function getTierThresholds_() {
  try {
    var sheet = getSheet_(SN.TIER_THRESHOLDS);
    if (!sheet) return {S:85,A:70,B:55,C:40,D:0};
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return {S:85,A:70,B:55,C:40,D:0};
    var result = {S:85,A:70,B:55,C:40,D:0};
    for (var i = 1; i < data.length; i++) {
      var tier = String(data[i][0]||'').trim().toUpperCase();
      var val  = parseFloat(data[i][1])||0;
      if (result.hasOwnProperty(tier)) result[tier] = val;
    }
    return result;
  } catch(e) {
    return {S:85,A:70,B:55,C:40,D:0};
  }
}

function saveTierThresholds(params) {
  try {
    var t = params.thresholds || params;
    var sheet = getOrCreateSheet_(SN.TIER_THRESHOLDS);
    sheet.clearContents();
    var rows = [['TIER','MIN_SCORE']];
    ['S','A','B','C','D'].forEach(function(tier){
      rows.push([tier, t[tier]!==undefined ? t[tier] : {S:85,A:70,B:55,C:40,D:0}[tier]]);
    });
    sheet.getRange(1,1,rows.length,2).setValues(rows);
    sheet.getRange(1,1,1,2).setFontWeight('bold').setBackground('#6366f1').setFontColor('#ffffff');
    sheet.setColumnWidth(1,80);
    sheet.setColumnWidth(2,120);
    return { ok: true };
  } catch(e) {
    return { ok: false, error: e.message };
  }
}

function getTierRulesData_() {
  var sheet = getSheet_(SN.TIER_RULES);
  if (!sheet) return getDefaultTierRules_().rules;
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return getDefaultTierRules_().rules;

  var headers = data[0].map(function(h){ return String(h||'').trim().toUpperCase(); });
  var mIdx  = headers.indexOf('METRIC');
  var kIdx  = headers.indexOf('KEY');
  var wIdx  = headers.indexOf('WEIGHT');
  var tiers = ['S','A','B','C','D'];
  var tIdx  = tiers.map(function(t){ return headers.indexOf(t); });

  var rules = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var label  = String(row[mIdx]||'').trim();
    if (!label) continue;
    var metric = kIdx >= 0 ? String(row[kIdx]||'').trim().toLowerCase() : label.toLowerCase().replace(/\s+/g,'_');
    var obj = { metric: metric, label: label, weight: parseFloat(row[wIdx])||0, thresholds:{} };
    tiers.forEach(function(t, ti) {
      obj.thresholds[t] = parseFloat(row[tIdx[ti]])||0;
    });
    rules.push(obj);
  }
  return rules.length ? rules : getDefaultTierRules_().rules;
}

function getTierAlertsData_() {
  var sheet = getSheet_(SN.TIER_ALERTS);
  if (!sheet) return getDefaultTierRules_().alerts;
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return getDefaultTierRules_().alerts;

  var headers = data[0].map(function(h){
    return String(h||'').trim().toLowerCase().replace(/%/g,'').replace(/[\s\-]/g,'_');
  });
  var tiers = ['S','A','B','C','D'];
  var alerts = {};
  for (var i = 1; i < data.length; i++) {
    var row  = data[i];
    var tier = String(row[0]||'').trim().toUpperCase();
    if (tiers.indexOf(tier) < 0) continue;
    alerts[tier] = {};
    for (var j = 1; j < headers.length; j++) {
      if (headers[j]) alerts[tier][headers[j]] = parseFloat(row[j])||0;
    }
  }
  alerts._columns = headers.slice(1).filter(function(h){ return !!h; });
  return Object.keys(alerts).length > 1 ? alerts : getDefaultTierRules_().alerts;
}

function _saveTierAlertsSheet_(alerts, columns) {

function normalizeColKey_(str) {
  // Normalize NFC trước để xử lý cả NFD (Google Sheets đôi khi lưu NFD)
  if (str.normalize) str = str.normalize('NFC');
  var from = 'àáảãạăắằẳẵặâấầẩẫậèéẻẽẹêếềểễệìíỉĩịòóỏõọôốồổỗộơớờởỡợùúủũụưứừửữựỳýỷỹỵđÀÁẢÃẠĂẮẰẲẴẶÂẤẦẨẪẬÈÉẺẼẸÊẾỀỂỄỆÌÍỈĨỊÒÓỎÕỌÔỐỒỔỖỘƠỚỜỞỠỢÙÚỦŨỤƯỨỪỬỮỰỲÝỶỸỴĐ';
  var to   = 'aaaaaaaaaaaaaaaaaeeeeeeeeeeeiiiiioooooooooooooooooouuuuuuuuuuuyyyyydAAAAAAAAAAAAAAAAAEEEEEEEEEEEIIIIIOOOOOOOOOOOOOOOOOOUUUUUUUUUUUYYYYYD';
  var result = '';
  for (var i = 0; i < str.length; i++) {
    var c = str[i];
    var idx = from.indexOf(c);
    result += (idx >= 0) ? to[idx] : c;
  }
  return result.toLowerCase().replace(/[^a-z0-9]+/g,'_').replace(/^_+|_+$/g,'');
}

function getDataColumns() {
  try {
    var sheet = getSheet_(SN.DATA);
    if (!sheet) return { ok: true, columns: [] };
    var lastCol = sheet.getLastColumn();
    if (lastCol < 1) return { ok: true, columns: [] };
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];

    // Alias: normalized header → canonical key (nhất quán với getData)
    var ALIAS = {
      wl_1:'wl', wl_m1:'wl', wl_x_1:'wl',
      margin_pct:'margin', ty_le_margin:'margin',
      peak_ccu:'ccu', dinh_ccu:'ccu',
      tong_km_nap_va_cac_km_khac_tru_hoa_hong_huy_km:'km_nap',
      khoa_gian_lan:'khoa',
      user_gian_lan_khoa:'user_gian',
      nru_spam_neu_co:'nru_spam',
      huy_khuyen_mai:'huy_km'
    };
    // Các key meta không phải metric — skip bất kể header
    var SKIP_KEYS = {brand:1,ty_gia:1,date:1,ngay:1,thang:1,nam:1,
                     ty_gia_vnd:1,ngay_1:1,thang_1:1,nam_1:1};

    var seen = {};
    var cols = [];
    headers.forEach(function(h, idx) {
      if (idx < 6) return; // bỏ qua cột A-F (BRAND, TỶ GIÁ, DATE, NGÀY, THÁNG, NĂM)
      var raw = String(h||'').trim();
      if (!raw) return;
      var normKey = normalizeColKey_(raw);
      if (!normKey) return;
      if (SKIP_KEYS[normKey]) return;
      var key = ALIAS[normKey] || normKey; // dùng canonical key
      if (seen[key]) return;
      seen[key] = true;
      cols.push({ key: key, name: raw, colIndex: idx });
    });
    // Fallback: nếu dynamic scan fail (NFD/encoding) → thêm km_nap từ COL hardcoded
    if (!seen['km_nap'] && COL.km_nap !== undefined) {
      cols.push({ key: 'km_nap', name: 'KM Nạp', colIndex: COL.km_nap });
    }
    return { ok: true, columns: cols };
  } catch(e) {
    return { ok: false, error: e.message };
  }
}

      {key:'tong_mua2', idx:COL.tong_mua2},
      {key:'tong_ban2', idx:COL.tong_ban2},
      {key:'nru_spam',  idx:COL.nru_spam},
      {key:'user_gian', idx:COL.user_gian}
    ].filter(function(dc){ return dc.idx !== undefined; });

    // Các cột đặc biệt cần xử lý riêng
    var CCU_IDX = COL.ccu; // peak, không sum
    var BRAND_IDX = COL.brand;
    var DATE_IDX  = COL.date;

    var curr = {}, prev = {}, allBrandIds = {};

    function agg(obj, row) {
      var id = String(row[BRAND_IDX]||'').trim();
      if (!id) return;
      if (!obj[id]) obj[id] = {};
      var b = obj[id];
      dynCols.forEach(function(dc) {
        var cell = row[dc.idx];
        // Bỏ qua cell rỗng/null — giữ null để phân biệt "không có data" vs "giá trị 0"
        if (cell === '' || cell === null || cell === undefined) return;
        var val = n(cell);
        if (dc.idx === CCU_IDX) {
          if (b[dc.key] == null || val > b[dc.key]) b[dc.key] = val; // peak
        } else {
          b[dc.key] = (b[dc.key] == null ? 0 : b[dc.key]) + val;
        }
      });
    }

    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      if (!String(row[BRAND_IDX]||'').trim()) continue;

      var rawDate = row[DATE_IDX];
      var dateStr = '';
      if (rawDate instanceof Date) {
        dateStr = Utilities.formatDate(rawDate, 'Asia/Ho_Chi_Minh', 'yyyy-MM-dd');
      } else if (typeof rawDate === 'number') {
        var d = new Date((rawDate - 25569) * 86400 * 1000);
        dateStr = Utilities.formatDate(d, 'Asia/Ho_Chi_Minh', 'yyyy-MM-dd');
      } else {
        dateStr = String(rawDate||'').trim();
      }
      if (!dateStr) continue;

      var inCurr = (!sFrom || dateStr >= sFrom) && (!sTo   || dateStr <= sTo);
      var inPrev = (!sFromP|| dateStr >= sFromP) && (!sToP || dateStr <= sToP);

      if (inCurr) agg(curr, row);
      if (inPrev) agg(prev, row);
      var bid_ = String(row[BRAND_IDX]||'').trim();
      if (bid_) allBrandIds[bid_] = 1;
    }

    function isMonthMode_(from, to) {
      if (params && params.rangeMode) return params.rangeMode === 'month';
      if (!from||!to) return false;
      if (from.substring(0,7)!==to.substring(0,7)) return false;
      return parseInt(from.substring(8,10))===1;
    }
    if (isMonthMode_(sFrom, sTo)) {
      var auCurr = getAuMonthly_(sFrom.substring(0,7));
      Object.keys(auCurr).forEach(function(bid){if(curr[bid]) curr[bid].au=auCurr[bid];});
    }
    if (isMonthMode_(sFromP, sToP)) {
      var auPrev2 = getAuMonthly_(sFromP.substring(0,7));
      Object.keys(auPrev2).forEach(function(bid){if(prev[bid]) prev[bid].au=auPrev2[bid];});
    }
    // Tính margin per brand từ wl/to (không sum % từ cột Q)
    Object.keys(curr).forEach(function(bid){
      var b = curr[bid];
      b.margin = b.to ? Math.round(b.wl / b.to * 10000) / 100 : 0;
    });
    Object.keys(prev).forEach(function(bid){
      var b = prev[bid];
      b.margin = b.to ? Math.round(b.wl / b.to * 10000) / 100 : 0;
    });
    var allBrandList = Object.keys(allBrandIds);
    var result = { ok: true, curr: curr, prev: prev, allBrandIds: allBrandList };

    try {
      var nowStr = Utilities.formatDate(new Date(), 'Asia/Ho_Chi_Minh', 'yyyy-MM-dd');
      var yestDate = new Date(new Date().getTime() - 86400000);
      var yestStr = Utilities.formatDate(yestDate, 'Asia/Ho_Chi_Minh', 'yyyy-MM-dd');
      var isRecent = (sTo === nowStr || sTo === yestStr || sFrom === nowStr || sFrom === yestStr);
      if (!isRecent) {
        var json = JSON.stringify(result);
        if (json.length < 100000) {
          CacheService.getScriptCache().put(cacheKey, json, 1800);
        }
      }
    } catch(e) {}

    return result;
  } catch(e) {
    return { ok: false, error: e.message };
  }
}

function getTrend(params) {
  try {
    var brandId = params.brandId;
    var metric  = params.metric;
    var period  = params.period || 'day';
    var nPts    = params.n || 14;

    // Cache 30 phút theo brandId + metric + period
    var cacheKey = 'bpd_trend_v1_' + getCacheBust_() + '_' + brandId + '_' + metric + '_' + period + '_' + nPts;
    try {
      var cached = CacheService.getScriptCache().get(cacheKey);
      if (cached) return JSON.parse(cached);
    } catch(ce) {}

    var sheet = getSheet_(SN.DATA);
    if (!sheet) return { ok: false, error: 'No DATA sheet' };
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return { ok: true, points: [] };

    var nowGmt7 = new Date(new Date().getTime() + 7*3600000);
    var endDate = new Date(nowGmt7);
    endDate.setUTCDate(endDate.getUTCDate() - 1);

    var startDate = new Date(endDate);
    if (period === 'day') {
      startDate.setUTCDate(startDate.getUTCDate() - (nPts - 1));
    } else if (period === 'week') {
      startDate.setUTCDate(startDate.getUTCDate() - (nPts * 7 - 1));
    } else if (period === 'month') {
      startDate.setUTCMonth(startDate.getUTCMonth() - (nPts - 1));
      startDate.setUTCDate(1);
    }

    var pad = function(x){ return x < 10 ? '0'+x : String(x); };
    var toISO = function(d){ return d.getUTCFullYear()+'-'+pad(d.getUTCMonth()+1)+'-'+pad(d.getUTCDate()); };

    var sFrom = toISO(startDate);
    var sTo   = toISO(endDate);

    // Dùng COL hardcoded — schema sheet cố định, không cần dynamic scan
    var metricColIdx = COL[metric] !== undefined ? COL[metric] : -1;
    var toColIdx     = COL.to;
    var wlColIdx     = COL.wl;
    var nruColIdx    = COL.nru;
    var fpuColIdx    = COL.fpu || -1;
    var npuColIdx    = COL.npu || -1;

    if (metricColIdx < 0 && metric !== 'fpu_npu' && metric !== 'margin') {
      return { ok: true, points: [] };
    }

    var agg = {};

    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      if (String(row[COL.brand]||'').trim() !== brandId) continue;

      var rawDate = row[COL.date];
      var dateStr = '';
      if (rawDate instanceof Date) {
        dateStr = Utilities.formatDate(rawDate, 'Asia/Ho_Chi_Minh', 'yyyy-MM-dd');
      } else if (typeof rawDate === 'number') {
        dateStr = Utilities.formatDate(new Date((rawDate-25569)*86400000), 'Asia/Ho_Chi_Minh', 'yyyy-MM-dd');
      } else {
        dateStr = String(rawDate||'').trim();
      }
      if (!dateStr || dateStr < sFrom || dateStr > sTo) continue;

      var groupKey = dateStr;
      if (period === 'week') {
        // Nhóm theo tuần bắt đầu từ sFrom, mỗi tuần 7 ngày
        var dMs = new Date(dateStr+'T00:00:00Z').getTime();
        var sMs = new Date(sFrom+'T00:00:00Z').getTime();
        var weekIdx = Math.floor((dMs - sMs) / (7*86400000));
        var wStartMs = sMs + weekIdx * 7 * 86400000;
        var wEndMs   = wStartMs + 6 * 86400000;
        var wS = new Date(wStartMs); var wE = new Date(wEndMs);
        var p2 = function(x){ return x<10?'0'+x:String(x); };
        groupKey = p2(wS.getUTCDate())+'/'+p2(wS.getUTCMonth()+1)+'-'+p2(wE.getUTCDate())+'/'+p2(wE.getUTCMonth()+1);
      } else if (period === 'month') {
        groupKey = dateStr.substring(0, 7);
      }

      if (!agg[groupKey]) agg[groupKey] = {sum:0, nru:0, fpu:0, npu:0, to:0, wl:0, cnt:0};
      var g = agg[groupKey];
      g.cnt++;
      if (nruColIdx >= 0) g.nru += n(row[nruColIdx]);
      if (fpuColIdx >= 0) g.fpu += n(row[fpuColIdx]);
      if (npuColIdx >= 0) g.npu += n(row[npuColIdx]);
      g.to  += n(row[toColIdx]);
      g.wl  += n(row[wlColIdx]);
      if (metricColIdx >= 0) g.sum += n(row[metricColIdx]);
    }

    // AU monthly: dùng AU_MONTHLY thay vì sum daily
    if (metric === 'au' && period === 'month') {
      var auKeys2 = Object.keys(agg).sort();
      var auPoints2 = auKeys2.map(function(k) {
        var auMap2 = getAuMonthly_(k);
        var auVal2 = auMap2[brandId] || 0;
        var p2 = k.split('-');
        var lbl2 = 'T'+parseInt(p2[1])+'/'+p2[0].substring(2);
        return { label: lbl2, value: auVal2 };
      });
      return { ok: true, points: auPoints2 };
    }

    var keys = Object.keys(agg).sort();
    var points = keys.map(function(k) {
      var g = agg[k];
      var val = 0;
      if (metric === 'fpu_npu') val = g.npu > 0 ? Math.round(g.fpu/g.npu*10000)/100 : 0;
      else if (metric === 'margin') val = g.to > 0 ? Math.round(g.wl/g.to*10000)/100 : 0;
      else val = Math.round(g.sum);

      var lbl = k; // day và week: groupKey đã là label
      if (period === 'day') lbl = k.substring(8)+'/'+k.substring(5,7);
      else if (period === 'month') { var p=k.split('-'); lbl='T'+parseInt(p[1])+'/'+p[0].substring(2); }

      return { label: lbl, value: val };
    });

    var trendResult = { ok: true, points: points };
    try {
      var json = JSON.stringify(trendResult);
      if (json.length < 100000) CacheService.getScriptCache().put(cacheKey, json, 1800);
    } catch(ce2) {}
    return trendResult;
  } catch(e) {
    return { ok: false, error: e.message };
  }
}

// ADVANCED WARNINGS — tính Streak + Drawdown cho tất cả brands
// Trả về: { brandId: { metric: {streak, drawdown, severity, lastValue, direction} } }
// params: { lookback (số ngày), metrics [array keys], thresholds {metric:{streak:{med,high,critical}, drawdown:{...}}} }
// ═══════════════════════════════════════════════════════════════
function getAllBrandsAdvWarnings(params) {
  try {
    var lookback = parseInt(params.lookback) || 30;
    var metrics  = params.metrics  || ['margin','au','wl','to'];
    var thresholds = params.thresholds || {};

    // Cache key theo lookback + metrics (thresholds không ảnh hưởng raw data nên cache riêng)
    var cacheKey = 'bpd:adv_warn:v1:' + lookback + ':' + metrics.sort().join(',');
    var cache = CacheService.getScriptCache();
    var cached = null;
    try { cached = cache.get(cacheKey); } catch(ce){}
    var rawData = null;
    if (cached) {
      try { rawData = JSON.parse(cached); } catch(ce){}
    }

    if (!rawData) {
      // Build raw time series từ sheet DATA
      var sheet = getSheet_(SN.DATA);
      if (!sheet) return { ok: true, warnings: {} };
      var data = sheet.getDataRange().getValues();
      if (data.length <= 1) return { ok: true, warnings: {} };

      // Tính date range: endDate = hôm qua GMT+7, startDate = endDate - (lookback-1)
      var nowGmt7 = new Date(new Date().getTime() + 7*3600000);
      var endDate = new Date(nowGmt7);
      endDate.setUTCDate(endDate.getUTCDate() - 1);
      var startDate = new Date(endDate);
      startDate.setUTCDate(startDate.getUTCDate() - (lookback - 1));
      var pad = function(x){ return x < 10 ? '0'+x : String(x); };
      var toISO = function(d){ return d.getUTCFullYear()+'-'+pad(d.getUTCMonth()+1)+'-'+pad(d.getUTCDate()); };
      var sFrom = toISO(startDate);
      var sTo   = toISO(endDate);

      // Gom theo brand -> date -> {values by metric}
      // Key metric cần ánh xạ sang COL
      rawData = {}; // { brandId: { 'yyyy-MM-dd': { raw:{nru,fpu,to,wl,...}, hasData:true } } }

      for (var i = 1; i < data.length; i++) {
        var row = data[i];
        var brandId = String(row[COL.brand]||'').trim();
        if (!brandId) continue;

        var rawDate = row[COL.date];
        var dateStr = '';
        if (rawDate instanceof Date) {
          dateStr = Utilities.formatDate(rawDate, 'Asia/Ho_Chi_Minh', 'yyyy-MM-dd');
        } else if (typeof rawDate === 'number') {
          dateStr = Utilities.formatDate(new Date((rawDate-25569)*86400000), 'Asia/Ho_Chi_Minh', 'yyyy-MM-dd');
        } else {
          dateStr = String(rawDate||'').trim();
        }
        if (!dateStr || dateStr < sFrom || dateStr > sTo) continue;

        if (!rawData[brandId]) rawData[brandId] = {};
        if (!rawData[brandId][dateStr]) {
          rawData[brandId][dateStr] = {
            nru:0, fpu:0, fau:0, npu:0, nau:0, pu:0, ccu:0,
            au:0, to:0, wl:0,
            tong_mua:0, tong_ban:0, mua_ban:0,
            km_nap:0, comm:0, khoa:0
          };
        }
        var d = rawData[brandId][dateStr];
        d.nru      += n(row[COL.nru]);
        d.fpu      += n(row[COL.fpu]);
        d.fau      += n(row[COL.fau]);
        d.npu      += n(row[COL.npu]);
        d.nau      += n(row[COL.nau]);
        d.pu       += n(row[COL.pu]);
        d.au       += n(row[COL.au]);
        d.to       += n(row[COL.to]);
        d.wl       += n(row[COL.wl]);
        d.tong_mua += n(row[COL.tong_mua]);
        d.tong_ban += n(row[COL.tong_ban]);
        d.mua_ban  += n(row[COL.mua_ban]);
        d.km_nap   += n(row[COL.km_nap]);
        d.comm     += n(row[COL.comm]);
        d.khoa     += n(row[COL.khoa]);
        var ccuV = n(row[COL.ccu]);
        if (ccuV > d.ccu) d.ccu = ccuV;
      }

      // Cache raw data 10 phút
      try {
        var rawJson = JSON.stringify(rawData);
        if (rawJson.length < 900000) cache.put(cacheKey, rawJson, 600);
      } catch(ce2){}
    }

    // Compute warnings per brand × metric
    var result = {};
    Object.keys(rawData).forEach(function(brandId){
      var datesMap = rawData[brandId];
      var dates = Object.keys(datesMap).sort(); // ascending
      if (dates.length < 2) return;

      var brandRes = {};
      metrics.forEach(function(mkey){
        var series = dates.map(function(dt){
          var d = datesMap[dt];
          if (mkey === 'margin') {
            return d.to > 0 ? (d.wl / d.to * 100) : 0;
          } else if (mkey === 'fpu_npu') {
            return d.nru > 0 ? (d.fpu / d.nru * 100) : 0;
          } else {
            return d[mkey] !== undefined ? d[mkey] : 0;
          }
        });

        var th = thresholds[mkey] || {direction:'down', streak:{med:3,high:5,critical:7}, drawdown:{med:20,high:35,critical:50}};
        var direction = th.direction || 'down';

        // Streak: số ngày liên tiếp ở cuối series mà giá trị thay đổi theo direction
        var streakDown = 0, streakUp = 0;
        for (var k = series.length - 1; k > 0; k--) {
          if (series[k] < series[k-1]) streakDown++;
          else break;
        }
        for (var k2 = series.length - 1; k2 > 0; k2--) {
          if (series[k2] > series[k2-1]) streakUp++;
          else break;
        }
        var streak = 0;
        if (direction === 'down') streak = streakDown;
        else if (direction === 'up') streak = streakUp;
        else streak = Math.max(streakDown, streakUp); // both

        // Drawdown: (max-min)/max × 100 trong window. Cần peak trước trough.
        var peak = series[0], trough = series[0], maxDD = 0, curPeak = series[0];
        for (var j = 0; j < series.length; j++) {
          var v = series[j];
          if (direction === 'down') {
            if (v > curPeak) curPeak = v;
            if (curPeak > 0) {
              var dd = (curPeak - v) / curPeak * 100;
              if (dd > maxDD) maxDD = dd;
            }
          } else if (direction === 'up') {
            if (v < curPeak) curPeak = v; // track trough
            if (curPeak !== 0) {
              var ddu = (v - curPeak) / Math.abs(curPeak) * 100;
              if (ddu > maxDD) maxDD = ddu;
            }
          } else { // both
            if (v > peak) peak = v;
            if (v < trough) trough = v;
            if (peak > 0) {
              var ddb = (peak - trough) / peak * 100;
              if (ddb > maxDD) maxDD = ddb;
            }
          }
        }

        // Severity: stable/medium/high/critical
        var streakSev = 'stable';
        if (streak >= (th.streak.critical||7)) streakSev = 'critical';
        else if (streak >= (th.streak.high||5)) streakSev = 'high';
        else if (streak >= (th.streak.med||3)) streakSev = 'medium';

        var ddSev = 'stable';
        if (maxDD >= (th.drawdown.critical||50)) ddSev = 'critical';
        else if (maxDD >= (th.drawdown.high||35)) ddSev = 'high';
        else if (maxDD >= (th.drawdown.med||20)) ddSev = 'medium';

        // Overall severity = worst of streak/dd
        var sevOrder = {stable:0, medium:1, high:2, critical:3};
        var overallSev = sevOrder[streakSev] >= sevOrder[ddSev] ? streakSev : ddSev;

        brandRes[mkey] = {
          streak: streak,
          drawdown: Math.round(maxDD*10)/10,
          streakSev: streakSev,
          ddSev: ddSev,
          severity: overallSev,
          lastValue: series[series.length-1],
          direction: direction,
          dataPoints: series.length,
          series: series.map(function(v){ return Math.round(v*100)/100; }) // sparkline data
        };
      });

      result[brandId] = brandRes;
    });

    return { ok: true, warnings: result, lookback: lookback };
  } catch(e) {
    Logger.log('getAllBrandsAdvWarnings error: ' + e.message);
    return { ok: false, error: e.message };
  }
}


function getAuMonthly_(yearMonth) {
  try {
    var sheet = getSheet_(SN.AU_MONTHLY);
    if (!sheet) return {};
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return {};
    var headers = data[0];
    var colIdx = -1;
    var monthNames = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
    for (var c = 1; c < headers.length; c++) {
      var h = String(headers[c]||'').trim();
      var parts = h.split(' ');
      if (parts.length === 2) {
        var mIdx = monthNames.indexOf(parts[0]) + 1;
        var yr = parseInt(parts[1]);
        var ym = yr + '-' + (mIdx < 10 ? '0'+mIdx : String(mIdx));
        if (ym === yearMonth) { colIdx = c; break; }
      }
    }
    if (colIdx < 0) return {};
    var result = {};
    for (var i = 1; i < data.length; i++) {
      var brandId = String(data[i][0]||'').trim();
      if (!brandId) continue;
      result[brandId] = parseFloat(data[i][colIdx])||0;
    }
    return result;
  } catch(e) { return {}; }
}

// Severity map — phân loại action thành 4 mức độ
// low    — hoạt động bình thường (login/logout)
// medium — thay đổi nhưng an toàn (thêm user/brand)
// high   — thay đổi nhạy cảm (xóa brand, sửa IP, reset Yubikey, sửa role)
// critical — cực kỳ quan trọng (xóa user, nhường quyền Owner)
// @FROZEN: batch=1 date=2026-04-22 reason="Severity 4 mức đã duyệt bởi sếp"
var SEVERITY_MAP_ = {
  'Đăng nhập'                      : 'low',
  'Đăng xuất'                      : 'low',
  'Đăng xuất tự động (inactivity)' : 'low',
  'THÊM user'                      : 'medium',
  'THÊM brand phụ trách'           : 'medium',
  'SỬA IP whitelist'               : 'medium',
  'SỬA role user'                  : 'high',
  'XÓA brand phụ trách'            : 'high',
  'XÓA IP whitelist'               : 'high',
  'RESET Yubikey'                  : 'high',
  'XÓA user'                       : 'critical',
  'NHƯỜNG quyền Owner'             : 'critical'
};

// @FROZEN: batch=1 date=2026-04-22 reason="Auto-classify action via keyword fallback"
function getSeverity_(action) {
  if (!action) return 'low';
  // Match chính xác trước
  if (SEVERITY_MAP_[action]) return SEVERITY_MAP_[action];
  // Fallback: match theo keyword
  var a = String(action).toLowerCase();
  if (a.indexOf('xóa user') >= 0 || a.indexOf('nhường') >= 0) return 'critical';
  if (a.indexOf('xóa') >= 0 || a.indexOf('reset') >= 0 || a.indexOf('sửa role') >= 0) return 'high';
  if (a.indexOf('thêm') >= 0 || a.indexOf('sửa') >= 0) return 'medium';
  return 'low';
}

// @FROZEN: batch=1 date=2026-04-22 reason="Schema 9 cột + auto-migrate 6->9, backward compat"
function logActivity_(action, email, brand, period, detail, ip, device, severity) {
  try {
    var sheet = getOrCreateSheet_(SN.ACTIVITY_LOG);
    var lastRow = sheet.getLastRow();
    var lastCol = sheet.getLastColumn();

    // Schema v2: 9 cột. Schema v1: 6 cột (cũ).
    // Auto-migrate: nếu sheet < 9 cột, thêm header cho cột mới.
    if (lastRow === 0) {
      // Sheet rỗng hoàn toàn → tạo header đầy đủ 9 cột
      sheet.appendRow(['timestamp','email','action','brand','period','detail','ip','device','severity']);
      sheet.getRange(1,1,1,9).setFontWeight('bold');
    } else if (lastCol < 9) {
      // Sheet đã có data với schema cũ → chỉ thêm header cho cột thiếu
      var newHeaders = ['ip','device','severity'];
      // lastCol có thể = 6 (data cũ) hoặc 7-8 (trường hợp lỗi giữa chừng)
      for (var c = lastCol + 1; c <= 9; c++) {
        sheet.getRange(1, c).setValue(newHeaders[c - 7]).setFontWeight('bold');
      }
    }

    // Auto-compute severity nếu frontend không truyền
    var sev = severity || getSeverity_(action);

    var ts = Utilities.formatDate(new Date(), 'Asia/Ho_Chi_Minh', 'dd/MM/yyyy HH:mm:ss');
    sheet.appendRow([
      ts,
      email||'',
      action||'',
      brand||'',
      period||'',
      detail||'',
      ip||'',
      device||'',
      sev
    ]);
  } catch(e) {}
}

    var numRows = lastRow - startRow + 1;
    // Đọc tối đa 9 cột — nếu sheet chỉ có 6 cột (data cũ), 3 cột cuối sẽ = ''
    var colsToRead = Math.min(9, Math.max(6, lastCol));
    var data = sheet.getRange(startRow, 1, numRows, colsToRead).getValues();
    var logs = [];
    for (var i = data.length - 1; i >= 0; i--) {
      var row = data[i];
      if (!row[0]) continue;
      // Format timestamp an toàn — có thể là Date object hoặc string
      var tsStr = '';
      try {
        if (row[0] instanceof Date) {
          tsStr = Utilities.formatDate(row[0], 'Asia/Ho_Chi_Minh', 'dd/MM/yyyy HH:mm:ss');
        } else {
          tsStr = String(row[0]);
        }
      } catch(te) { tsStr = String(row[0]); }

      var action = String(row[2] || '');
      // Nếu severity rỗng (log cũ hoặc chưa có), auto-compute từ action
      var severity = String(row[8] || '') || getSeverity_(action);

      logs.push({
        ts      : tsStr,
        email   : String(row[1] || ''),
        action  : action,
        brand   : String(row[3] || ''),
        period  : String(row[4] || ''),
        detail  : String(row[5] || ''),
        ip      : String(row[6] || ''),
        device  : String(row[7] || ''),
        severity: severity
      });
    }
    return { ok: true, logs: logs, total: total, limit: limit, timezone: 'GMT+7' };
  } catch(e) {
    return { ok: false, error: e.message, logs: [], total: 0, timezone: 'GMT+7' };
  }
}

// @FROZEN: batch=3 date=2026-04-22 reason="Online list, role ưu tiên từ ACCESS_LIST"
function getActiveSessions(params) {
  try {
    var token = getFirebaseAccessToken_(getServiceAccountKey_());
    var url = 'https://firestore.googleapis.com/v1/projects/' + FIREBASE_PROJECT_ID +
              '/databases/(default)/documents/sessions?pageSize=200';
    var resp = UrlFetchApp.fetch(url, {
      headers: { 'Authorization': 'Bearer ' + token },
      muteHttpExceptions: true
    });
    if (resp.getResponseCode() !== 200) return { ok: true, sessions: [] };
    var docs = JSON.parse(resp.getContentText()).documents || [];
    var now = Date.now();

    // Map email → session mới nhất (exp lớn nhất)
    var byEmail = {};
    docs.forEach(function(doc) {
      var f = doc.fields || {};
      var email = f.email && f.email.stringValue ? f.email.stringValue.trim() : '';
      var role  = f.role  && f.role.stringValue  ? f.role.stringValue : 'Viewer';
      var exp   = f.exp   && (f.exp.integerValue || f.exp.doubleValue)
                  ? parseInt(f.exp.integerValue || f.exp.doubleValue) : 0;
      var lastSeen = f.lastSeen && (f.lastSeen.integerValue || f.lastSeen.doubleValue)
                  ? parseInt(f.lastSeen.integerValue || f.lastSeen.doubleValue) : 0;
      if (!email || !exp) return;

      // exp lưu dạng milliseconds (Date.now() + TTL)
      // Nếu exp < 1e12 thì đang ở dạng seconds → chuyển sang ms
      if (exp < 1e12) exp = exp * 1000;
      if (lastSeen && lastSeen < 1e12) lastSeen = lastSeen * 1000;

      // Giữ session có exp lớn nhất cho mỗi email
      if (!byEmail[email] || exp > byEmail[email].exp) {
        byEmail[email] = { email: email, role: role, exp: exp, lastSeen: lastSeen };
      }
    });

    // Đọc role thật từ ACCESS_LIST — ưu tiên hơn session (session có thể lưu role cũ)
    try {
      var allUsers = getUsers();
      Object.keys(byEmail).forEach(function(em) {
        var found = allUsers.find(function(u){ return (u.email||'').trim().toLowerCase() === em.toLowerCase(); });
        if (found && found.role) byEmail[em].role = found.role;
      });
    } catch(re) { Logger.log('getActiveSessions role sync err: ' + re.message); }

    var ONLINE_THRESHOLD_MS = 90 * 1000; // 90 giây không heartbeat = offline
    var sessions = [];
    Object.keys(byEmail).forEach(function(email) {
      var s = byEmail[email];
      // Fallback cho session cũ chưa có lastSeen: dùng exp-TTL để ước tính
      var tokenValid = s.exp > now;
      var isOnline = tokenValid && s.lastSeen && (now - s.lastSeen < ONLINE_THRESHOLD_MS);
      var minsLeft = tokenValid ? Math.round((s.exp - now) / 60000) : 0;
      sessions.push({
        email    : s.email,
        role     : s.role,
        exp      : s.exp,
        lastSeen : s.lastSeen,
        isOnline : isOnline,
        minsLeft : minsLeft
      });
    });

    sessions.sort(function(a,b){
      return (b.isOnline?1:0)-(a.isOnline?1:0) || b.exp-a.exp;
    });
    return { ok: true, sessions: sessions };
  } catch(e) {
    return { ok: false, error: e.message, sessions: [] };
  }
}

// @FROZEN: batch=3 date=2026-04-22 reason="Firestore runQuery delete all sessions by email"
function deleteUserSessions(params) {
  try {
    var email = params && params.email ? params.email.trim().toLowerCase() : '';
    if (!email) return { ok: false, error: 'Missing email' };

    var token = getFirebaseAccessToken_(getServiceAccountKey_());
    var projectPath = 'projects/' + FIREBASE_PROJECT_ID + '/databases/(default)';
    var apiBase = 'https://firestore.googleapis.com/v1/' + projectPath;

    // Dùng Firestore runQuery để filter thẳng theo email — không bị giới hạn pageSize
    var queryBody = {
      structuredQuery: {
        from: [{ collectionId: 'sessions' }],
        where: {
          fieldFilter: {
            field: { fieldPath: 'email' },
            op: 'EQUAL',
            value: { stringValue: email }
          }
        }
      }
    };

    var queryResp = UrlFetchApp.fetch(apiBase + '/documents:runQuery', {
      method: 'POST',
      headers: { 'Authorization': 'Bearer ' + token, 'Content-Type': 'application/json' },
      payload: JSON.stringify(queryBody),
      muteHttpExceptions: true
    });

    var deleted = 0;
    if (queryResp.getResponseCode() === 200) {
      var results = JSON.parse(queryResp.getContentText());
      if (!Array.isArray(results)) results = [results];
      results.forEach(function(item) {
        if (!item.document || !item.document.name) return;
        try {
          var docName = item.document.name;
          // docName dạng: projects/.../databases/(default)/documents/sessions/{id}
          var deleteUrl = docName.indexOf('https://') === 0
            ? docName
            : 'https://firestore.googleapis.com/v1/' + docName;
          UrlFetchApp.fetch(deleteUrl, {
            method: 'DELETE',
            headers: { 'Authorization': 'Bearer ' + token },
            muteHttpExceptions: true
          });
          deleted++;
        } catch(de) { Logger.log('Delete session err: ' + de.message); }
      });
    }

    Logger.log('deleteUserSessions: xóa ' + deleted + ' sessions của ' + email);
    return { ok: true, deleted: deleted };
  } catch(e) {
    Logger.log('deleteUserSessions error: ' + e.message);
    return { ok: false, error: e.message };
  }
}

// @FROZEN: batch=3 date=2026-04-22 reason="Retention 90 ngày / 5000 rows"
var ACTIVITY_LOG_MAX_DAYS = 90;   // giữ 90 ngày gần nhất
var ACTIVITY_LOG_MAX_ROWS = 5000; // giữ tối đa 5000 dòng

// @FROZEN: batch=3 date=2026-04-22 reason="Trigger Thứ Hai 3SA — dọn ACTIVITY_LOG + setupCleanupTrigger"
function cleanupActivityLog() {
  try {
    var sheet = getSheet_(SN.ACTIVITY_LOG);
    if (!sheet) return;
    var lastRow = sheet.getLastRow();
    if (lastRow <= 1) return; // chỉ có header hoặc trống

    var now = new Date();
    var cutoffDate = new Date(now.getTime() - ACTIVITY_LOG_MAX_DAYS * 24 * 3600 * 1000);

    var data = sheet.getRange(2, 1, lastRow - 1, 1).getValues(); // chỉ lấy cột timestamp
    var rowsToDelete = [];

    for (var i = 0; i < data.length; i++) {
      var raw = data[i][0];
      if (!raw) { rowsToDelete.push(i + 2); continue; }
      // Parse timestamp format dd/MM/yyyy HH:mm:ss
      var ts = null;
      try {
        if (raw instanceof Date) {
          ts = raw;
        } else {
          var parts = String(raw).trim().split(' ');
          var dateParts = parts[0].split('/');
          var timeParts = (parts[1] || '00:00:00').split(':');
          ts = new Date(
            parseInt(dateParts[2]), parseInt(dateParts[1]) - 1, parseInt(dateParts[0]),
            parseInt(timeParts[0]), parseInt(timeParts[1]), parseInt(timeParts[2] || 0)
          );
        }
      } catch(pe) {}
      if (!ts || ts < cutoffDate) rowsToDelete.push(i + 2);
    }

    // Xóa từ dưới lên để không lệch index
    rowsToDelete.sort(function(a,b){return b-a;});
    var deleted = 0;
    for (var j = 0; j < rowsToDelete.length; j++) {
      sheet.deleteRow(rowsToDelete[j]);
      deleted++;
    }

    // Kiểm tra thêm giới hạn MAX_ROWS sau khi xóa theo ngày
    var newLastRow = sheet.getLastRow();
    if (newLastRow - 1 > ACTIVITY_LOG_MAX_ROWS) {
      var excess = newLastRow - 1 - ACTIVITY_LOG_MAX_ROWS;
      sheet.deleteRows(2, excess); // xóa các dòng cũ nhất (đầu sheet)
      deleted += excess;
    }

    Logger.log('cleanupActivityLog: đã xóa ' + deleted + ' dòng. Còn lại: ' + (sheet.getLastRow()-1) + ' dòng.');
    return { ok: true, deleted: deleted, remaining: sheet.getLastRow() - 1 };
  } catch(e) {
    Logger.log('cleanupActivityLog error: ' + e.message);
    return { ok: false, error: e.message };
  }
}

function cleanupExpiredSessions() {
  try {
    var token = getFirebaseAccessToken_(getServiceAccountKey_());
    var baseUrl = 'https://firestore.googleapis.com/v1/projects/' + FIREBASE_PROJECT_ID
                + '/databases/(default)/documents/sessions?pageSize=200';

      headers: { 'Authorization': 'Bearer ' + token },
      muteHttpExceptions: true
    });
    if (resp.getResponseCode() !== 200) {
      Logger.log('cleanupExpiredSessions: fetch failed ' + resp.getResponseCode());
      return;
    }
    var docs = JSON.parse(resp.getContentText()).documents || [];
    var now = Date.now();
    var deleted = 0;
    docs.forEach(function(doc) {
      var f = doc.fields || {};
      var exp = f.exp ? parseInt(f.exp.integerValue || f.exp.doubleValue || 0) : 0;

            headers: { 'Authorization': 'Bearer ' + token },

        } catch(de) {}
      }
    });
    Logger.log('cleanupExpiredSessions: xóa ' + deleted + ' sessions hết hạn. Còn lại: ' + (docs.length - deleted));
  } catch(e) {
    Logger.log('cleanupExpiredSessions error: ' + e.message);
  }

function getKpiMonthly(params) {
  try {
    var metric = params.metric;
    var months = params.months || [];
    if (metric === 'au') {
      return months.map(function(ym) {
        var auMap = getAuMonthly_(ym);
        var total = Object.keys(auMap).reduce(function(s,k){return s+(auMap[k]||0);},0);
        return { month: ym, value: total };
      });
    }
    var sheet = getSheet_(SN.DATA);
    if (!sheet) return months.map(function(m){return {month:m,value:0};});
    var data = sheet.getDataRange().getValues();
    var tz7 = 7*3600000;
    function toYM(raw) {
      var ms;
      if (raw instanceof Date) ms=raw.getTime()+tz7;
      else if (typeof raw==='number') ms=(raw-25569)*86400000+tz7;
      else return '';
      var d=new Date(ms), mo=d.getUTCMonth()+1;
      return d.getUTCFullYear()+'-'+(mo<10?'0'+mo:String(mo));
    }
    var colIdx = COL[metric];
    if (colIdx===undefined) return months.map(function(m){return {month:m,value:0};});
    var totals={};
    months.forEach(function(m){totals[m]=0;});
    for (var i=1;i<data.length;i++) {
      var row=data[i];
      if (!String(row[COL.brand]||'').trim()) continue;
      var ym=toYM(row[COL.date]);
      if (!ym||!totals.hasOwnProperty(ym)) continue;
      totals[ym]+=n(row[colIdx]);
    }
    return months.map(function(m){return {month:m,value:Math.round(totals[m])};});
  } catch(e) { return []; }
}

function n(v) {
  var x = parseFloat(v);
  return isNaN(x) ? 0 : x;
}

      if (verResp.getResponseCode() === 200) {
        versionCount = (JSON.parse(verResp.getContentText()).versions || []).length;
      }
      // Đọc source để đếm dòng + lấy CACHE_VER thật
      var srcResp = UrlFetchApp.fetch(
        'https://script.googleapis.com/v1/projects/' + scriptId + '/content',
        { headers: { 'Authorization': 'Bearer ' + token }, muteHttpExceptions: true }
      );
      if (srcResp.getResponseCode() === 200) {
        var files = JSON.parse(srcResp.getContentText()).files || [];
        files.forEach(function(f) {
          if (f.type === 'SERVER_JS') {
            codeLines = (f.source.match(/\n/g) || []).length + 1;
            var m = f.source.match(/var\s+CACHE_VER\s*=\s*'([^']+)'/);
            if (m) cacheVerReal = m[1];
          }
          if (f.name === 'index') {
            htmlLines = (f.source.match(/\n/g) || []).length + 1;
          }
        });
      }
    } catch(e2) {}

    return {
      ok           : true,
      deployUrl    : deployUrl,
      spreadsheetId: SPREADSHEET_ID,
      scriptId     : scriptId,
      gasVersion   : versionCount + ' / 200',
      htmlLines    : htmlLines ? htmlLines.toLocaleString() + ' dòng' : '—',
      codeLines    : codeLines ? codeLines.toLocaleString() + ' dòng' : '—',
      cacheVer     : cacheVerReal + ' · active',
      timestamp    : Utilities.formatDate(new Date(), 'Asia/Ho_Chi_Minh', 'dd/MM/yyyy HH:mm:ss')
    };
  } catch(e) {
    return { ok: false, error: e.message };
  }
}

function handleApi_(e) {
  var action = e.parameter.action || '';
  var email  = (e.parameter.email || '').toLowerCase().trim();
  var p      = {};
  try { p = JSON.parse(decodeURIComponent(e.parameter.p || '{}')); } catch(err) {}

  if (!email) return jsonOut_({ok:false, error:'Missing email'});
  var users = getUsers();
  var found = users.find(function(u){ return u.email === email; });
  if (!found) return jsonOut_({ok:false, error:'Unauthorized'});

  var result;
  try {
    switch(action) {
      case 'loadAll':
        var pcMaxUsers = 0;
        try {
          var wToken = getFirebaseAccessToken_(getServiceAccountKey_());
          syncWebauthnFromFirestore_(wToken);
          try { CacheService.getScriptCache().remove('bpd_users'); } catch(ce) {}
          users = getUsers();

          var cfgResp = UrlFetchApp.fetch(cfgUrl,{headers:{'Authorization':'Bearer '+wToken},muteHttpExceptions:true});
          if(cfgResp.getResponseCode()===200){
            var cfgDoc = JSON.parse(cfgResp.getContentText());
            if(cfgDoc.fields&&cfgDoc.fields.max_users){
              pcMaxUsers = parseInt(cfgDoc.fields.max_users.integerValue||cfgDoc.fields.max_users.doubleValue||0)||0;
            }
          }
          Logger.log('loadAll: maxUsers='+pcMaxUsers);
        } catch(we) {
          Logger.log('loadAll error: ' + we.message);
        }
        result = {
          ok: true,
          tierRules:   getTierRules(),
          appConfig:   getAppConfig(),
          users:       users,
          brandLevels: getBrandLevels(),
          currentUser: found,
          maxUsers:    pcMaxUsers
        };
        break;
      case 'getData':       result = getData(p);            break;
      case 'bustCache':     result = bustCache();           break;
      case 'getNotes':      result = {ok:true, notes: getNotes()}; break;
      case 'saveNote':      result = saveNote(p);           break;
      case 'deleteNote':    result = deleteNote(p);         break;
      case 'saveUsers':     result = saveUsersSync_(p.users||p); break;
      case 'saveConfig':    result = savePCConfig_(p); break;
      case 'saveUsersSync':  result = saveUsersSync_(p.users||p); break;
      case 'deleteUser':     result = deleteUser_(p);             break;
      case 'resetUserYubikey': result = resetUserYubikey_(p);     break;
      case 'saveUserIp':     result = saveUserIp_(p);             break;

      case 'saveAppConfig': result = saveAppConfig(p);      break;
      case 'getAdvancedAlert':  result = getAdvancedAlert(p);  break;
      case 'saveAdvancedAlert': result = saveAdvancedAlert(p); break;
      case 'getAllBrandsAdvWarnings': result = getAllBrandsAdvWarnings(p); break;
      case 'saveBrandLevels': result = saveBrandLevels(p.levels||p); break;
      case 'getGiaiTrinh':   result = getGiaiTrinh(p); break;
      case 'saveGiaiTrinh':  result = saveGiaiTrinh(p); break;
      case 'delGiaiTrinh':   result = delGiaiTrinh(p); break;
      case 'logActivity':       logActivity(p); result = {ok:true}; break;
      case 'getActivityLog':    result = getActivityLog(p); break;
      case 'getActiveSessions':    result = getActiveSessions(p); break;
      case 'updateHeartbeat':      result = updateHeartbeat(p); break;

function getFirestoreEmails_(token) {
  try {
    var url = 'https://firestore.googleapis.com/v1/projects/' + FIREBASE_PROJECT_ID +
              '/databases/(default)/documents/access_list?pageSize=200';
    var resp = UrlFetchApp.fetch(url, {
      headers: { 'Authorization': 'Bearer ' + token },
      muteHttpExceptions: true
    });
    if (resp.getResponseCode() !== 200) return [];
    var data = JSON.parse(resp.getContentText());
    var docs = data.documents || [];
    var emails = [];
    docs.forEach(function(doc) {
      var f = doc.fields || {};
      var email = f.email && f.email.stringValue ? f.email.stringValue.trim().toLowerCase() : '';
      if (email && email.indexOf('@') >= 0) emails.push(email);
    });
    return emails;
  } catch(e) {
    Logger.log('getFirestoreEmails_ error: ' + e.message);
    return [];
  }
}

function syncAccessListToFirestore_(token) {
  var sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(SN.USERS);
  if(!sheet) return;
  var data = sheet.getDataRange().getValues();
  if(data.length <= 1) return;
  var h = data[0].map(function(x){ return String(x).trim().toLowerCase(); });
  var idx = {name:h.indexOf('name'),email:h.indexOf('email'),role:h.indexOf('role'),color:h.indexOf('color'),level:h.indexOf('level'),brands:h.indexOf('brands'),webauthn:h.indexOf('webauthn_id'),ip:h.indexOf('ip_whitelist')};
  var ipDocs = [];
  for(var i=1;i<data.length;i++){
    var row=data[i];
    var email=String(row[idx.email]||'').trim().toLowerCase();
    if(!email||!email.includes('@')) continue;
    var docId=email.replace(/[.@]/g,'_');
    var webauthnId=idx.webauthn>=0?String(row[idx.webauthn]||'').trim():'';
    var ip=idx.ip>=0?String(row[idx.ip]||'').trim():'';
    var fields={
      email:{stringValue:email},name:{stringValue:String(row[idx.name]||'').trim()},
      role:{stringValue:String(row[idx.role]||'Viewer').trim()},color:{stringValue:String(row[idx.color]||'#94a3b8').trim()},
      level:{stringValue:String(row[idx.level]||'').trim()},brands:{stringValue:String(row[idx.brands]||'').trim()},
      ip_whitelist:{stringValue:ip}
    };
    var maskFields = ['email','name','role','color','level','brands','ip_whitelist'];
    var maskQuery = maskFields.map(function(f){ return 'updateMask.fieldPaths='+encodeURIComponent(f); }).join('&');
    var patchUrl = 'https://firestore.googleapis.com/v1/projects/'+FIREBASE_PROJECT_ID+'/databases/(default)/documents/access_list/'+docId+'?'+maskQuery;
    UrlFetchApp.fetch(patchUrl,{method:'PATCH',contentType:'application/json',headers:{'Authorization':'Bearer '+token},payload:JSON.stringify({fields:fields}),muteHttpExceptions:true});
    if(ip) ipDocs.push({ip:ip,label:'',email:email});
  }
  ipDocs.forEach(function(d){
    firestorePatch__(token,'ip_whitelist/'+d.ip.replace(/\./g,'_'),{fields:{ip:{stringValue:d.ip},label:{stringValue:d.label},email:{stringValue:d.email}}});
  });
}

function getGiaiTrinh_(params) {
  try {
    var sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(SN.GIAI_TRINH);
    if (!sheet || sheet.getLastRow() <= 1) return { ok: true, data: [] };
    var data = sheet.getDataRange().getValues();
    var h = data[0].map(function(x){ return String(x).trim().toLowerCase(); });
    var bIdx = h.indexOf('brand_id'), uIdx = h.indexOf('url');
    if (bIdx < 0 || uIdx < 0) return { ok: true, data: [] };
    var result = [];
    for (var i = 1; i < data.length; i++) {
      var brand = String(data[i][bIdx]||'').trim();
      if (!brand) continue;
      result.push({ brand_id: brand, url: String(data[i][uIdx]||'').trim() });
    }
    return { ok: true, data: result };
  } catch(e) { return { ok: false, error: e.message, data: [] }; }

function saveGiaiTrinh_(params) {
  try {
    var brand_id = (params.brand_id||'').trim();
    var url = (params.url||'').trim();
    var by = (params.updated_by||'').trim();
    if (!brand_id) return { ok: false, error: 'Missing brand_id' };
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName(SN.GIAI_TRINH);
    if (!sheet) {
      sheet = ss.insertSheet(SN.GIAI_TRINH);
      sheet.appendRow(['brand_id','url','updated_at','updated_by']);
      sheet.getRange(1,1,1,4).setFontWeight('bold').setBackground('#6366f1').setFontColor('#ffffff');
    }
    var now = Utilities.formatDate(new Date(),'Asia/Ho_Chi_Minh','dd/MM/yyyy HH:mm:ss');
    var lastRow = sheet.getLastRow();
    if (lastRow <= 1) {
      // Chỉ có header hoặc trống — append thẳng
      sheet.appendRow([brand_id, url, now, by]);
      return { ok: true };
    }
    // Đọc cột brand_id để tìm row cần update (1 lần đọc duy nhất)
    var brandCol = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
    for (var i = 0; i < brandCol.length; i++) {
      if (String(brandCol[i][0]||'').trim() === brand_id) {
        // Update cả row bằng 1 lần setValues
        sheet.getRange(i+2, 1, 1, 4).setValues([[brand_id, url, now, by]]);
        return { ok: true };
      }
    }
    // Không tìm thấy — thêm mới
    sheet.appendRow([brand_id, url, now, by]);
    return { ok: true };
  } catch(e) { return { ok: false, error: e.message }; }
}

function delGiaiTrinh_(params) {
  try {
    var brand_id = (params.brand_id||'').trim();
    if (!brand_id) return { ok: false, error: 'Missing brand_id' };
    var sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(SN.GIAI_TRINH);

      }

}

function deleteUser_(params) {
  try {
    var email = (params.email || '').trim().toLowerCase();
    if (!email) return { ok: false, error: 'Missing email' };

    var sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(SN.USERS);
    if (!sheet) return { ok: false, error: 'Sheet khong ton tai' };
    var data = sheet.getDataRange().getValues();
    var h = data[0].map(function(x){ return String(x).trim().toLowerCase(); });
    var emailIdx = h.indexOf('email');
    var ipIdx    = h.indexOf('ip_whitelist');
    if (emailIdx < 0) return { ok: false, error: 'Thieu cot email' };

    var userIp = '';
    var deleted = false;
    for (var i = data.length - 1; i >= 1; i--) {
      if (String(data[i][emailIdx] || '').trim().toLowerCase() === email) {
        if (ipIdx >= 0) userIp = String(data[i][ipIdx] || '').trim();
        sheet.deleteRow(i + 1);
        deleted = true;
        break;
      }
    }
    if (!deleted) return { ok: false, error: 'Khong tim thay user: ' + email };

    var token = getFirebaseAccessToken_(getServiceAccountKey_());
    var docId = email.replace(/[.@]/g, '_');
    UrlFetchApp.fetch(
      'https://firestore.googleapis.com/v1/projects/' + FIREBASE_PROJECT_ID + '/databases/(default)/documents/access_list/' + docId,
      { method: 'DELETE', headers: { 'Authorization': 'Bearer ' + token }, muteHttpExceptions: true }
    );
    if (userIp) {
      UrlFetchApp.fetch(
        'https://firestore.googleapis.com/v1/projects/' + FIREBASE_PROJECT_ID + '/databases/(default)/documents/ip_whitelist/' + userIp.replace(/\./g, '_'),
        { method: 'DELETE', headers: { 'Authorization': 'Bearer ' + token }, muteHttpExceptions: true }
      );
    }
    try {
      var sessUrl = 'https://firestore.googleapis.com/v1/projects/' + FIREBASE_PROJECT_ID +
                    '/databases/(default)/documents/sessions?pageSize=200';
      var sessResp = UrlFetchApp.fetch(sessUrl, {
        headers: { 'Authorization': 'Bearer ' + token },
        muteHttpExceptions: true
      });
      if (sessResp.getResponseCode() === 200) {
        var sessData = JSON.parse(sessResp.getContentText());
        var docs = sessData.documents || [];
        docs.forEach(function(doc) {
          var f = doc.fields || {};
          var sessEmail = f.email && f.email.stringValue ? f.email.stringValue.trim().toLowerCase() : '';
          if (sessEmail === email) {
            UrlFetchApp.fetch(
              'https://firestore.googleapis.com/v1/' + doc.name,
              { method: 'DELETE', headers: { 'Authorization': 'Bearer ' + token }, muteHttpExceptions: true }
            );
            Logger.log('Deleted session for: ' + email);
          }
        });
      }
    } catch(sessErr) {
      Logger.log('Delete session error: ' + sessErr.message);
    }
    try { CacheService.getScriptCache().remove('bpd_users'); } catch(e2) {}
    return { ok: true };
  } catch(e) {
    return { ok: false, error: e.message };
  }
}

    var data = sheet.getDataRange().getValues();
    var h = data[0].map(function(x){ return String(x).trim().toLowerCase(); });
    var emailIdx    = h.indexOf('email');
    var webauthnIdx = h.indexOf('webauthn_id');
    if (emailIdx < 0 || webauthnIdx < 0) return { ok: false, error: 'Thieu cot' };

    var found = false;
    for (var i = 1; i < data.length; i++) {

        sheet.getRange(i + 1, webauthnIdx + 1).setValue('');
        found = true;
        break;
      }
    }
    if (!found) return { ok: false, error: 'Khong tim thay user: ' + email };

    var token = getFirebaseAccessToken_(getServiceAccountKey_());
    var docId = email.replace(/[.@]/g, '_');
    var url = 'https://firestore.googleapis.com/v1/projects/' + FIREBASE_PROJECT_ID +
              '/databases/(default)/documents/access_list/' + docId +

    UrlFetchApp.fetch(url, {
      method: 'PATCH',
      contentType: 'application/json',
      headers: { 'Authorization': 'Bearer ' + token },

    });

    return { ok: false, error: e.message };
  }
}

function saveUserIp_(params) {
  try {
    var email    = (params.email    || '').trim().toLowerCase();
    var ip       = (params.ip       || '').trim();
    if (!email) return { ok: false, error: 'Missing email' };

    var sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(SN.USERS);
    if (!sheet) return { ok: false, error: 'Sheet khong ton tai' };
    var data = sheet.getDataRange().getValues();
    var h = data[0].map(function(x){ return String(x).trim().toLowerCase(); });
    var emailIdx = h.indexOf('email');
    var ipIdx    = h.indexOf('ip_whitelist');
    if (emailIdx < 0) return { ok: false, error: 'Thieu cot email' };

    var found = false;
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][emailIdx] || '').trim().toLowerCase() === email) {
        if (ipIdx    >= 0) sheet.getRange(i + 1, ipIdx    + 1).setValue(ip);
        found = true;
        break;
      }
    }
    if (!found) return { ok: false, error: 'Khong tim thay user: ' + email };

    var token = getFirebaseAccessToken_(getServiceAccountKey_());
    var docId = email.replace(/[.@]/g, '_');
    var url = 'https://firestore.googleapis.com/v1/projects/' + FIREBASE_PROJECT_ID +
              '/databases/(default)/documents/access_list/' + docId +
              '?updateMask.fieldPaths=ip_whitelist';
    UrlFetchApp.fetch(url, {
      method: 'PATCH',
      contentType: 'application/json',
      headers: { 'Authorization': 'Bearer ' + token },
      payload: JSON.stringify({ fields: {
        ip_whitelist: { stringValue: ip },
      }}),
      muteHttpExceptions: true
    });
    if (ip) {
      // Thêm/cập nhật IP mới vào Firestore
      firestorePatch__(token, 'ip_whitelist/' + ip.replace(/\./g, '_'), {
        fields: {
          ip:    { stringValue: ip },
          email: { stringValue: email }
        }
      });
    } else {
      // IP bị xóa (để trống) → xóa tất cả document ip_whitelist của email này
      try {
        var ipQueryBody = {
          structuredQuery: {
            from: [{ collectionId: 'ip_whitelist' }],
            where: { fieldFilter: { field: { fieldPath: 'email' }, op: 'EQUAL', value: { stringValue: email } } }
          }
        };
        var ipQueryResp = UrlFetchApp.fetch(
          'https://firestore.googleapis.com/v1/projects/' + FIREBASE_PROJECT_ID + '/databases/(default)/documents:runQuery',
          { method: 'POST', headers: { 'Authorization': 'Bearer ' + token, 'Content-Type': 'application/json' },
            payload: JSON.stringify(ipQueryBody), muteHttpExceptions: true }
        );
        var ipResults = JSON.parse(ipQueryResp.getContentText());
        if (!Array.isArray(ipResults)) ipResults = [ipResults];
        ipResults.forEach(function(item) {
          if (!item.document || !item.document.name) return;
          var delUrl = 'https://firestore.googleapis.com/v1/' + item.document.name;
          UrlFetchApp.fetch(delUrl, { method: 'DELETE', headers: { 'Authorization': 'Bearer ' + token }, muteHttpExceptions: true });
          Logger.log('Deleted ip_whitelist doc: ' + item.document.name.split('/').pop());
        });
      } catch(ipErr) { Logger.log('Delete ip_whitelist err: ' + ipErr.message); }
    }
    try { CacheService.getScriptCache().remove('bpd_users'); } catch(e2) {}
    return { ok: true };
  } catch(e) {
    return { ok: false, error: e.message };
  }
}

// ─────────────────────────────────────────────
//  ANALYTICS DATA — tổng hợp sẵn cho tab Analytics
//  Nhẹ hơn getData() vì không cần raw từng dòng
//  Params: { dateFrom, dateTo, dateFromP, dateToP }
//  Returns: { ok, totals, totals_prev, tier_dist }
// ─────────────────────────────────────────────
function getAnalyticsData(params) {
  try {
    var sFrom  = params && params.dateFrom  ? params.dateFrom  : null;
    var sTo    = params && params.dateTo    ? params.dateTo    : null;
    var sFromP = params && params.dateFromP ? params.dateFromP : null;
    var sToP   = params && params.dateToP   ? params.dateToP   : null;

    // Cache 30 phút (không cache nếu date là hôm nay/hôm qua)
    var cacheKey = 'bpd_analytics_v1_' + getCacheBust_() + '_' + (sFrom||'') + '_' + (sTo||'') + '_' + (sFromP||'') + '_' + (sToP||'');
    try {
      var nowGmt7 = new Date(new Date().getTime() + 7*3600000);
      var yestDate = new Date(nowGmt7); yestDate.setUTCDate(yestDate.getUTCDate()-1);
      var nowStr  = Utilities.formatDate(nowGmt7,  'Asia/Ho_Chi_Minh', 'yyyy-MM-dd');
      var yestStr = Utilities.formatDate(yestDate, 'Asia/Ho_Chi_Minh', 'yyyy-MM-dd');
      var isRecent = (sTo===nowStr || sTo===yestStr || sFrom===nowStr || sFrom===yestStr);
      if (!isRecent) {
        var cached = CacheService.getScriptCache().get(cacheKey);
        if (cached) return JSON.parse(cached);
      }
    } catch(ce) {}

    var sheet = getSheet_(SN.DATA);
    if (!sheet) return { ok: false, error: 'No DATA sheet' };
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return { ok: true, totals: {}, totals_prev: {}, brand_count: 0 };

    // Dùng COL hardcoded — không phụ thuộc tên header
    var dynCols = [
      {key:'nru',       idx:COL.nru},
      {key:'fpu',       idx:COL.fpu},
      {key:'fau',       idx:COL.fau},
      {key:'npu',       idx:COL.npu},
      {key:'nau',       idx:COL.nau},
      {key:'pu',        idx:COL.pu},
      {key:'ccu',       idx:COL.ccu},
      {key:'au',        idx:COL.au},
      {key:'to',        idx:COL.to},
      {key:'wl',        idx:COL.wl},
      {key:'sl_mua',    idx:COL.sl_mua},
      {key:'tong_mua',  idx:COL.tong_mua},
      {key:'sl_ban',    idx:COL.sl_ban},
      {key:'tong_ban',  idx:COL.tong_ban},
      {key:'mua_ban',   idx:COL.mua_ban},
      {key:'km_nap',    idx:COL.km_nap},
      {key:'comm',      idx:COL.comm},
      {key:'khoa',      idx:COL.khoa},
      {key:'km_hoan',   idx:COL.km_hoan},
      {key:'km_khac',   idx:COL.km_khac},
      {key:'huy_km',    idx:COL.huy_km},
      {key:'tong_mua2', idx:COL.tong_mua2},
      {key:'tong_ban2', idx:COL.tong_ban2},
      {key:'nru_spam',  idx:COL.nru_spam},
      {key:'user_gian', idx:COL.user_gian}
    ].filter(function(dc){ return dc.idx !== undefined; });

    var BRAND_IDX = COL.brand;
    var DATE_IDX  = COL.date;
    var curr = {}, prev = {};
    var currBrandCount = {}, prevBrandCount = {};

    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var id = String(row[BRAND_IDX]||'').trim();
      if (!id) continue;

      var rawDate = row[DATE_IDX];
      var dateStr = '';
      if (rawDate instanceof Date) {
        dateStr = Utilities.formatDate(rawDate, 'Asia/Ho_Chi_Minh', 'yyyy-MM-dd');
      } else if (typeof rawDate === 'number') {
        dateStr = Utilities.formatDate(new Date((rawDate-25569)*86400000), 'Asia/Ho_Chi_Minh', 'yyyy-MM-dd');
      } else {
        dateStr = String(rawDate||'').trim();
      }
      if (!dateStr) continue;

      var inCurr = (!sFrom || dateStr >= sFrom) && (!sTo   || dateStr <= sTo);
      var inPrev = (!sFromP|| dateStr >= sFromP) && (!sToP || dateStr <= sToP);

      if (inCurr) {
        currBrandCount[id] = 1;
        dynCols.forEach(function(dc) {
          var val = n(row[dc.idx]);
          curr[dc.key] = (curr[dc.key] || 0) + val;
        });
      }
      if (inPrev) {
        prevBrandCount[id] = 1;
        dynCols.forEach(function(dc) {
          var val = n(row[dc.idx]);
          prev[dc.key] = (prev[dc.key] || 0) + val;
        });
      }
    }

    // Override AU từ AU_MONTHLY nếu là month mode
    function isMonthModeAn_(from, to) {
      if (!from || !to) return false;
      if (from.substring(0,7) !== to.substring(0,7)) return false;
      return parseInt(from.substring(8,10)) === 1;
    }
    if (isMonthModeAn_(sFrom, sTo)) {
      var auCurrMap = getAuMonthly_(sFrom.substring(0,7));
      var auCurrTotal = 0;
      Object.keys(auCurrMap).forEach(function(bid){ auCurrTotal += (auCurrMap[bid]||0); });
      curr.au = auCurrTotal; // luôn dùng AU_MONTHLY, kể cả khi = 0
    }
    if (isMonthModeAn_(sFromP, sToP)) {
      var auPrevMap = getAuMonthly_(sFromP.substring(0,7));
      var auPrevTotal = 0;
      Object.keys(auPrevMap).forEach(function(bid){ auPrevTotal += (auPrevMap[bid]||0); });
      prev.au = auPrevTotal;
    }

    // Tính margin từ wl/to
    curr.margin = curr.to ? Math.round(curr.wl / curr.to * 10000) / 100 : 0;
    prev.margin = prev.to ? Math.round(prev.wl / prev.to * 10000) / 100 : 0;

    var result = {
      ok          : true,
      totals      : curr,
      totals_prev : prev,
      brand_count : Object.keys(currBrandCount).length
    };
    // Lưu cache 30 phút
    try {
      var json = JSON.stringify(result);
      if (json.length < 100000) CacheService.getScriptCache().put(cacheKey, json, 1800);
    } catch(ce2) {}
    return result;
  } catch(e) {
    return { ok: false, error: e.message };
  }
}


// ── Lấy danh sách năm có data trong sheet ──
function getDataYears() {
  try {
    var sheet = getSheet_(SN.DATA);
    if (!sheet) return { ok: true, years: [] };
    var data = sheet.getDataRange().getValues();
    var seenYears = {};
    for (var i = 1; i < data.length; i++) {
      var rawDate = data[i][COL.date];
      var year = null;
      if (rawDate instanceof Date) {
        year = rawDate.getFullYear();
      } else if (typeof rawDate === 'number') {
        year = new Date((rawDate-25569)*86400000).getFullYear();
      } else if (typeof rawDate === 'string' && rawDate.length >= 4) {
        year = parseInt(rawDate.substring(0,4));
      }
      if (year && year > 2000) seenYears[year] = 1;
    }
    var years = Object.keys(seenYears).map(Number).sort(function(a,b){return b-a;}); // desc
    return { ok: true, years: years };
  } catch(e) {
    return { ok: false, error: e.message };
  }
}

function getAnalyticsTrendAll(params) {
  try {
    var metrics = params.metrics || ['au','to','wl','margin'];
    var mode    = params.mode    || 'day';
    var sFrom   = params.dateFrom || '';
    var sTo     = params.dateTo   || '';
    if (!sFrom || !sTo) return { ok: false, error: 'Missing date range' };

    // ── GAS Cache: tránh scan lại nếu đã có kết quả ──
    var cacheKey = 'trend_v3_'+mode+'_'+getCacheBust_()+'_'+metrics.sort().join('_')+'_'+sFrom+'_'+sTo;
    try {
      var cache = CacheService.getScriptCache();
      var cached = cache.get(cacheKey);
      if (cached) return JSON.parse(cached);
    } catch(ce) {}

    var pad = function(n){ return n<10?'0'+n:String(n); };
    var now7 = new Date(new Date().getTime() + 7*3600000);

    // ── Build time buckets ──
    var buckets = []; // [{key, label, from, to}]

    if (mode === 'month') {
      var endD    = new Date(sTo+'T00:00:00Z');
      var endYear = endD.getUTCFullYear();
      var endMon  = endD.getUTCMonth();
      for (var i=11; i>=0; i--) {
        var d = new Date(Date.UTC(endYear, endMon-i, 1));
        var ym = d.getUTCFullYear()+'-'+pad(d.getUTCMonth()+1);
        var mFrom = ym+'-01';
        var lastDay = new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth()+1, 0)).getUTCDate();
        var curYM = endYear+'-'+pad(endMon+1);
        var mTo = (ym===curYM) ? sTo : ym+'-'+pad(lastDay);
        buckets.push({ key:ym, label:'T'+(d.getUTCMonth()+1)+'/'+d.getUTCFullYear(), from:mFrom, to:mTo, isMonth:true, ym:ym });
      }
    } else if (mode === 'week') {
      var yestW = new Date(now7); yestW.setUTCDate(yestW.getUTCDate()-1);
      for (var w=3; w>=0; w--) {
        var wEnd = new Date(yestW); wEnd.setUTCDate(yestW.getUTCDate()-w*7);
        var wStart = new Date(wEnd); wStart.setUTCDate(wEnd.getUTCDate()-6);
        var wFrom = wStart.getUTCFullYear()+'-'+pad(wStart.getUTCMonth()+1)+'-'+pad(wStart.getUTCDate());
        var wTo   = wEnd.getUTCFullYear()+'-'+pad(wEnd.getUTCMonth()+1)+'-'+pad(wEnd.getUTCDate());
        var lbl = pad(wStart.getUTCDate())+'/'+pad(wStart.getUTCMonth()+1)+'–'+pad(wEnd.getUTCDate())+'/'+pad(wEnd.getUTCMonth()+1);
        buckets.push({ key:wFrom, label:lbl, from:wFrom, to:wTo });
      }
    } else {
      // day mode: từng ngày trong range
      var msF = new Date(sFrom+'T00:00:00Z').getTime();
      var msT = new Date(sTo+'T00:00:00Z').getTime();
      var days = Math.round((msT-msF)/86400000)+1;
      for (var day=0; day<days; day++) {
        var ts = msF+day*86400000;
        var dt = new Date(ts);
        var dStr = dt.getUTCFullYear()+'-'+pad(dt.getUTCMonth()+1)+'-'+pad(dt.getUTCDate());
        buckets.push({ key:dStr, label:pad(dt.getUTCDate())+'/'+pad(dt.getUTCMonth()+1), from:dStr, to:dStr });
      }
    }

    // ── Lấy AU_MONTHLY nếu cần (mode month và AU trong list) ──
    var auMonthlyCache = {};
    var needAuMonthly = (mode==='month' && metrics.indexOf('au')>=0);
    if (needAuMonthly) {
      // Đọc AU_MONTHLY sheet 1 lần duy nhất cho tất cả tháng cần
      try {
        var auSheet = getSheet_(SN.AU_MONTHLY);
        if (auSheet) {
          var auData = auSheet.getDataRange().getValues();
          var auHeaders = auData[0];
          var monthNames = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
          // Build map: ym → colIdx
          var ymColMap = {};
          for (var ac=1; ac<auHeaders.length; ac++) {
            var ah = String(auHeaders[ac]||'').trim().split(' ');
            if (ah.length===2) {
              var ami = monthNames.indexOf(ah[0])+1;
              var ayr = parseInt(ah[1]);
              if (ami>0) ymColMap[ayr+'-'+(ami<10?'0'+ami:String(ami))] = ac;
            }
          }
          // Tổng hợp AU cho từng tháng cần
          buckets.forEach(function(b){
            if (!b.ym) return;
            var colIdx = ymColMap[b.ym];
            if (colIdx===undefined) { auMonthlyCache[b.ym]=0; return; }
            var total = 0;
            for (var ar=1; ar<auData.length; ar++) {
              total += parseFloat(auData[ar][colIdx])||0;
            }
            auMonthlyCache[b.ym] = total;
          });
        }
      } catch(auErr) {}
    }

    // ── Đọc header sheet DATA 1 lần ──
    var sheet = getSheet_(SN.DATA);
    if (!sheet) return { ok: false, error: 'No DATA sheet' };
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return { ok: true, data: {} };

    var headers = data[0];
    // Dùng COL hardcoded — không phụ thuộc tên header
    var metricColMap = {};
    var needMargin = metrics.indexOf('margin') >= 0;
    metrics.forEach(function(mk){
      if (mk==='au' && needAuMonthly) return;
      if (mk==='margin') return;
      if (COL[mk] !== undefined) metricColMap[mk] = COL[mk];
    });
    if (needMargin) {
      metricColMap['to'] = COL.to;
      metricColMap['wl'] = COL.wl;
    }

    // ── Init totals per bucket per metric ──
    var totals = {}; // bucketKey → { metric → value, to, wl }
    buckets.forEach(function(b){ totals[b.key] = {}; });

    // ── Build date range nhanh để skip sớm ──
    var globalFrom = buckets[0].from;
    var globalTo   = buckets[buckets.length-1].to;

    // ── Build bucket lookup map để tìm nhanh O(1) cho day mode ──
    var bucketMap = {}; // dateStr/key → bucketKey
    if (mode === 'day') {
      buckets.forEach(function(b){ bucketMap[b.key] = b.key; });
    }

    // ── Scan DATA sheet 1 lần ──
    for (var i=1; i<data.length; i++) {
      var row = data[i];
      if (!String(row[COL.brand]||'').trim()) continue;
      var rawDate = row[COL.date];
      var dateStr = '';
      if (rawDate instanceof Date) {
        dateStr = Utilities.formatDate(rawDate, 'Asia/Ho_Chi_Minh', 'yyyy-MM-dd');
      } else if (typeof rawDate === 'number') {
        dateStr = Utilities.formatDate(new Date((rawDate-25569)*86400000), 'Asia/Ho_Chi_Minh', 'yyyy-MM-dd');
      } else {
        dateStr = String(rawDate||'').trim();
      }
      if (!dateStr) continue;
      // Skip sớm nếu ngoài global range
      if (dateStr < globalFrom || dateStr > globalTo) continue;

      // Tìm bucket
      var bucketKey = null;
      if (mode === 'day') {
        bucketKey = bucketMap[dateStr] || null;
      } else {
        for (var b=0; b<buckets.length; b++) {
          if (dateStr >= buckets[b].from && dateStr <= buckets[b].to) {
            bucketKey = buckets[b].key; break;
          }
        }
      }
      if (!bucketKey || !totals[bucketKey]) continue;

      // Cộng dồn từng metric
      var t = totals[bucketKey];
      var mKeys = Object.keys(metricColMap);
      for (var mk=0; mk<mKeys.length; mk++) {
        var k = mKeys[mk];
        t[k] = (t[k]||0) + n(row[metricColMap[k]]);
      }
    }

    // ── Build result ──
    var result = {};
    metrics.forEach(function(mk){ result[mk] = []; });

    buckets.forEach(function(b){
      var t = totals[b.key] || {};
      metrics.forEach(function(mk){
        var val = 0;
        if (mk==='au' && needAuMonthly && b.ym) {
          val = auMonthlyCache[b.ym] || 0;
        } else if (mk==='margin') {
          val = (t.to && t.to>0) ? Math.round(t.wl/t.to*10000)/100 : 0;
        } else {
          val = Math.round(t[mk]||0);
        }
        result[mk].push({ label: b.label, value: val });
      });
    });

    var finalResult = { ok: true, data: result };
    // Lưu cache 6 tiếng
    try {
      var cache2 = CacheService.getScriptCache();
      var str = JSON.stringify(finalResult);
      if (str.length < 100000) { // GAS cache limit 100KB
        cache2.put(cacheKey, str, 21600);
      }
    } catch(ce2) {}
    return finalResult;
  } catch(e) {
    return { ok: false, error: e.message };
  }
}
